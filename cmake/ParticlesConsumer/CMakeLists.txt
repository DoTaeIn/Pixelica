cmake_minimum_required(VERSION 3.21)
project(ParticlesConsumer LANGUAGES C)
if(NOT PARTICLES_COMPONENT)
  set(PARTICLES_COMPONENT particles)
endif()
find_package(Pixelica CONFIG REQUIRED COMPONENTS ${PARTICLES_COMPONENT})
if(PARTICLES_COMPONENT STREQUAL "particles" AND
   (TARGET Pixelica::core OR TARGET Pixelica::renderer2d OR TARGET Pixelica::audio))
  message(FATAL_ERROR "Portable particles unexpectedly imported a renderer or audio")
endif()
if(PARTICLES_COMPONENT STREQUAL "particles2d" AND TARGET Pixelica::core)
  message(FATAL_ERROR "2D particles unexpectedly imported the 3D renderer")
endif()
add_executable(particles_consumer main.c)
target_link_libraries(particles_consumer PRIVATE Pixelica::${PARTICLES_COMPONENT})
target_compile_definitions(particles_consumer PRIVATE "PARTICLES_COMPONENT_${PARTICLES_COMPONENT}=1")
enable_testing()
add_test(NAME particles_consumer COMMAND particles_consumer)
