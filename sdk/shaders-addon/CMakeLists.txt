cmake_minimum_required(VERSION 3.21)
project(PixelicaShaders VERSION 0.1.0 LANGUAGES C)
include(GNUInstallDirs)
option(PIXELICA_INSTALL "Install the optional shader module" ON)
option(PIXELICA_SHADERS_BUILD_EXAMPLE "Build the offscreen shader example" OFF)
find_package(Pixelica 0.1 CONFIG REQUIRED COMPONENTS common)
if(NOT TARGET pixelica_common)
  add_library(pixelica_common ALIAS Pixelica::common)
endif()
if(NOT TARGET pixelica_options)
  add_library(pixelica_options INTERFACE)
endif()
add_subdirectory(modules/shaders)
if(PIXELICA_SHADERS_BUILD_EXAMPLE)
  add_executable(hello_shaders examples/hello_shaders/main.c)
  target_link_libraries(hello_shaders PRIVATE Pixelica::shaders)
  enable_testing()
  add_test(NAME smoke_hello_shaders COMMAND hello_shaders)
endif()
